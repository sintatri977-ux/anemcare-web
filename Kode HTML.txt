                    <p class="text-slate-800 font-medium text-sm sm:text-base leading-snug mb-3.5">${q.text}</p>
                    <div class="grid grid-cols-3 gap-2">
                        ${optionsHTML}
                    </div>
                `;

                container.appendChild(card);
            });

            updateQuizProgress();
            lucide.createIcons();
        }

        }

        async function initCamera() {
            const videoEl = document.getElementById('camera-video');
            const fallbackEl = document.getElementById('camera-fallback');
            const errMsgEl = document.getElementById('camera-error-msg');

            stopCameraStream();

            if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                videoEl.classList.add('hidden');
                fallbackEl.classList.remove('hidden');
                fallbackEl.classList.add('flex');
                errMsgEl.textContent = 'Browser Anda tidak mendukung akses kamera langsung. Silakan gunakan Mode Sampel Simulasi atau Unggah Foto.';
                return;
            }

            try {
                const stream = await navigator.mediaDevices.getUserMedia({
                    video: { facingMode: APP_STATE.facingMode, width: { ideal: 1280 }, height: { ideal: 720 } },
                    audio: false
                });
                APP_STATE.cameraStream = stream;
                videoEl.srcObject = stream;
                videoEl.classList.remove('hidden');
                fallbackEl.classList.add('hidden');
                fallbackEl.classList.remove('flex');
            } catch (err) {
                videoEl.classList.add('hidden');
                fallbackEl.classList.remove('hidden');
                fallbackEl.classList.add('flex');
                errMsgEl.textContent = 'Izin kamera ditolak atau kamera tidak terdeteksi. Gunakan tab "Sampel Simulasi / Unggah" untuk melanjutkan.';
            }
        }

        }

        // Ekstraksi Warna Rata-rata di Area Tengah (ROI Konjungtiva) dari Video atau Gambar Unggahan
        function analyzeImageColorROI(sourceElement) {
            try {
                const canvas = document.getElementById('pixel-analysis-canvas');
                const ctx = canvas.getContext('2d');
                ctx.drawImage(sourceElement, 0, 0, canvas.width, canvas.height);

                // Ambil sampel area tengah (Region of Interest oval konjungtiva)
                const roiX = Math.floor(canvas.width * 0.35);
                const roiY = Math.floor(canvas.height * 0.42);
                const roiW = Math.floor(canvas.width * 0.30);
                const roiH = Math.floor(canvas.height * 0.18);

                const imgData = ctx.getImageData(roiX, roiY, roiW, roiH).data;
                let rSum = 0, gSum = 0, bSum = 0, count = 0;

                for (let i = 0; i < imgData.length; i += 4) {
                    rSum += imgData[i];
                    gSum += imgData[i + 1];
                    bSum += imgData[i + 2];
                    count++;
                }

                const rAvg = rSum / count;
                const gAvg = gSum / count;
                const bAvg = bSum / count;

                // Hitung Erythema / Redness Index pendekatan CIELAB a*
                const rednessDiff = Math.max(0, rAvg - ((gAvg + bAvg) / 2));
                const cielabA = parseFloat(Math.min(32, Math.max(4, (rednessDiff * 0.32) + 5)).toFixed(1));

                let riskScore = 25;
                let label = 'Merah Muda (Sedang)';
                if (cielabA >= 20) {
                    riskScore = 15;
                    label = 'Merah Segar (Normal)';
                } else if (cielabA >= 12) {
                    riskScore = 50;
                    label = 'Merah Muda Pucat (Sedang)';
                } else {
                    riskScore = 82;
                    label = 'Sangat Pucat (Tinggi)';
                }

                return { riskContribution: riskScore, cielabA, label };
            } catch (e) {
                return { riskContribution: 22, cielabA: 21.4, label: 'Merah Segar (Normal)' };
            }
        }

        function startConjunctivaScan() {
            const btnScan = document.getElementById('btn-start-scan');
            const laserEl = document.getElementById('scanner-laser-line');
            const hudEl = document.getElementById('scan-hud');
            const statusText = document.getElementById('scan-status-text');
            const percentText = document.getElementById('scan-percent-text');
            const targetOval = document.getElementById('target-oval');
            const resultBanner = document.getElementById('scan-result-banner');

            // Cek jika mode kamera tetapi kamera gagal aktif
            if (APP_STATE.scanMode === 'camera' && !APP_STATE.cameraStream) {
                showToast('Kamera tidak aktif. Beralih ke Sampel Simulasi atau izinkan kamera.', 'warning');
                return;
            }

            btnScan.disabled = true;
            btnScan.classList.add('opacity-70', 'cursor-not-allowed');
            laserEl.classList.remove('hidden');
            hudEl.classList.remove('hidden');
            resultBanner.classList.add('hidden');
            targetOval.classList.remove('border-dashed', 'border-emerald-400');
            targetOval.classList.add('border-solid', 'border-emerald-300', 'scale-105');

            let progress = 0;
            const steps = [
                { pct: 25, msg: 'Mendeteksi batas kelopak mata bawah...' },
                { pct: 55, msg: 'Mengekstrak spektrum RGB & CIELAB a*...' },
                { pct: 85, msg: 'Menghitung indeks perfusi kapiler...' },
                { pct: 100, msg: 'Analisis citra konjungtiva selesai!' }
            ];

            const interval = setInterval(() => {
                progress += 5;
                percentText.textContent = `${progress}%`;

                const currentStage = steps.find(s => progress <= s.pct);
                if (currentStage) statusText.textContent = currentStage.msg;

                if (progress >= 100) {
                    clearInterval(interval);

                    // Tentukan hasil berdasarkan sumber pemindaian
                    if (APP_STATE.scanMode === 'camera' && APP_STATE.cameraStream) {
                        const videoEl = document.getElementById('camera-video');
                        APP_STATE.scanResult = analyzeImageColorROI(videoEl);
                    } else if (APP_STATE.selectedSampleType === 'uploaded') {
                        const imgEl = document.getElementById('uploaded-eye-img');
                        APP_STATE.scanResult = analyzeImageColorROI(imgEl);
                    } else {
                        const preset = SAMPLE_PRESETS[APP_STATE.selectedSampleType] || SAMPLE_PRESETS.normal;
                        APP_STATE.scanResult = {
                            riskContribution: preset.riskScore,
                            cielabA: preset.cielabA,
                            label: preset.label
                        };
                    }

                    APP_STATE.scanCompleted = true;

                    // Update UI setelah scan selesai
                    setTimeout(() => {
                        laserEl.classList.add('hidden');
                        hudEl.classList.add('hidden');
                        targetOval.classList.remove('scale-105');
                        btnScan.disabled = false;
                        btnScan.classList.remove('opacity-70', 'cursor-not-allowed');

                        document.getElementById('scan-summary-detail').textContent = 
                            `Indeks Kemerahan CIELAB a*: ${APP_STATE.scanResult.cielabA} • Kategori: ${APP_STATE.scanResult.label}`;
                        resultBanner.classList.remove('hidden');

                        showToast('Pemindaian berhasil! Melanjutkan ke hasil skor risiko...', 'success');
                        
                        // Otomatis lanjut ke Layar 4 setelah 1.2 detik agar mulus
                        setTimeout(() => {
                            if (APP_STATE.currentScreen === 3) {
                                goToScreen(4);
                            }
                        }, 1200);
                    }, 350);
                }
            }, 90);
        }

        // LOGIKA LAYAR 4: SKOR RISIKO GABUNGAN & EDUKASI TTD
        // =========================================================================
        function calculateAndRenderFinalReport() {
            // 1. Hitung Skor Kuesioner (Maksimal 7 * 2 = 14)
            let rawQuizScore = 0;
            QUESTIONS.forEach(q => {
                rawQuizScore += (APP_STATE.answers[q.id] || 0);
            });
            const maxQuizScore = QUESTIONS.length * 2;
            const quizPercentage = (rawQuizScore / maxQuizScore) * 100;

            // 2. Ambil Skor Pemindaian Konjungtiva (Jika user belum klik pindai, pakai default sample)
            const scanPercentage = APP_STATE.scanResult.riskContribution;

            // 3. Gabungkan Skor (Bobot: 45% Kuesioner + 55% Pemindaian Konjungtiva)
            const totalRisk = Math.round((quizPercentage * 0.45) + (scanPercentage * 0.55));

            // 4. Tentukan Kategori Risiko
            let category = 'Risiko Rendah';
            let badgeClass = 'bg-emerald-100 text-emerald-800';
            let strokeColor = '#10B981';
            let headline = 'Indikasi Risiko Anemia Rendah';
            let recommendation = 'Hasil skrining menunjukkan gejala klinis minimal dan warna konjungtiva tampak segar. Pertahankan pola makan bergizi seimbang kaya zat besi dan konsumsi Tablet Tambah Darah (TTD) secara rutin 1 tablet per minggu sebagai pencegahan.';

            if (totalRisk >= 65) {
                category = 'Risiko Tinggi';
                badgeClass = 'bg-rose-100 text-rose-800';
                strokeColor = '#E11D48';
                headline = 'Terdeteksi Indikasi Risiko Anemia Tinggi';
                recommendation = 'Kombinasi keluhan gejala klinis dan tingkat kepucatan konjungtiva mengarah pada risiko anemia yang tinggi. Sangat disarankan segera memeriksakan kadar Hemoglobin (Hb) di Puskesmas atau dokter untuk mendapatkan terapi TTD yang tepat.';
            } else if (totalRisk >= 35) {
                category = 'Risiko Sedang';
                badgeClass = 'bg-amber-100 text-amber-800';
                strokeColor = '#F59E0B';
                headline = 'Terdeteksi Indikasi Risiko Anemia Sedang';
                recommendation = 'Terdapat beberapa tanda gejala awal anemia atau sedikit kepucatan pada konjungtiva bawah. Tingkatkan konsumsi makanan kaya zat besi dan Vitamin C, konsumsi TTD sesuai anjuran, serta lakukan pengecekan Hb berkala di fasilitas kesehatan.';
            }

            // 5. Render ke Elemen DOM
            document.getElementById('total-risk-score').textContent = `${totalRisk}%`;
            
            const badgeEl = document.getElementById('risk-category-badge');
            badgeEl.textContent = category;
            badgeEl.className = `mt-3 px-4 py-1.5 rounded-full text-xs sm:text-sm font-bold ${badgeClass}`;

            document.getElementById('risk-headline').textContent = headline;
            document.getElementById('risk-recommendation').textContent = recommendation;

            // Animasi Lingkaran Gauge (Keliling lingkaran r=50 adalah ~314.16)
            const circumference = 314.16;
            const offset = circumference - (totalRisk / 100) * circumference;
            const gaugeCircle = document.getElementById('risk-gauge-circle');
            gaugeCircle.style.stroke = strokeColor;
            setTimeout(() => {
                gaugeCircle.style.strokeDashoffset = offset;
            }, 100);

            // Rincian Sub-Skor
            document.getElementById('detail-quiz-score').textContent = `${rawQuizScore} / ${maxQuizScore}`;
            document.getElementById('detail-quiz-label').textContent = 
                rawQuizScore >= 9 ? 'Gejala Sering' : (rawQuizScore >= 4 ? 'Gejala Sedang' : 'Gejala Minimal');

            document.getElementById('detail-scan-index').textContent = `a*: ${APP_STATE.scanResult.cielabA}`;
            document.getElementById('detail-scan-label').textContent = APP_STATE.scanResult.label;
        }
